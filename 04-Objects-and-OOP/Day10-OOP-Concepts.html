1. Setter and Getter Methods


class Person {
    constructor(name) {
        this._name = name; // underscore is a naming convention, not true JavaScript privacy
    }

    // Getter method علشان نقرأ قيمة _name
    get name() {
        return this._name; // بنرجع قيمة _name
    }

    // Setter method علشان نعدل قيمة _name
    set name(newName) {
        this._name = newName; // بنعدل قيمة _name
    }
}

let person = new Person("Code Wave"); // بننشئ object باسم Code Wave
console.log(person.name); // بنستخدم Getter علشان نقرأ الاسم: Code Wave

person.name = "Merna"; // بنستخدم Setter علشان نعدل الاسم
console.log(person.name); // بنستخدم Getter علشان نقرأ الاسم الجديد: Merna

2. Relationships Between Classes

Association


class Teacher {
    constructor(name) {
        this.name = name; // بنعرف خاصية name
    }
}

class Student {
    constructor(name, teacher) {
        this.name = name; // بنعرف خاصية name
        this.teacher = teacher; // بنعرف خاصية teacher
    }
}

let teacher = new Teacher("Mr. Smith"); // بننشئ object من الكلاس Teacher
let student = new Student("John", teacher); // هنا الاسم "John"

console.log(student.teacher.name); // بنطبع اسم المدرس: Mr. Smith

Aggregation

Composition


class Engine {
    start() {
        return "Engine started"; // بنرجع رسالة لما المحرك يشتغل
    }
}

class Car {
    constructor() {
        this.engine = new Engine(); // بننشئ object من الكلاس Engine
    }

    start() {
        return `${this.engine.start()}, Car started`; // بنرجع رسالة لما العربية تتحرك
    }
}

let car = new Car(); // بننشئ object من الكلاس Car
console.log(car.start()); // بنشغل العربية: Engine started, Car started

3. Inheritance


class Animal {
    constructor(name) {
        this.name = name; // بنعرف خاصية name
    }

    speak() {
        return `${this.name} makes a noise.`; // بنرجع رسالة
    }
}

class Dog extends Animal {
    constructor(name, breed) {
        super(name); // نستدعي constructor الكلاس الأساسي
        this.breed = breed; // بنعرف خاصية breed
    }

    speak() {
        return `${this.name} barks.`; // بنرجع رسالة
    }
}

let dog = new Dog("Rex", "German Shepherd"); // هنا الاسم "Rex"
console.log(dog.speak()); // بنستدعي دالة speak: Rex barks.
